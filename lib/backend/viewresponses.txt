SELECT 
    q.question_id,
    q.question_text,
    q.category,
    q.question_type,
    CASE 
        WHEN q.question_type = 'single_choice' THEN
            (
                SELECT answer_text 
                FROM answers 
                WHERE question_id = q.question_id 
                AND answer_value = ur.answers[1]
                LIMIT 1
            )

        WHEN q.question_type = 'multi_choice' THEN
            (
                SELECT string_agg(answer_text, ', ') 
                FROM answers 
                WHERE question_id = q.question_id 
                AND answer_value = ANY(ur.answers)
            )

        WHEN q.question_type = 'text' THEN
            ur.answers[1]::text
    END AS answer_text,

    ur.answers AS raw_values
    
FROM user_response ur
JOIN questions q 
    ON ur.question_id = q.question_id

WHERE ur.user_id = 'b8a13133-8b32-4b37-a9cb-74ad18992b85'

ORDER BY q.question_id;
